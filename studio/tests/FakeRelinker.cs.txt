// A stand-in for relinker.exe, built with the .NET Framework compiler for the
// tests. It writes what the real relinker writes for a successful conversion.
// The same file also acts as the "converted game": a copy of it that isn't
// named relinker.exe behaves like a game being run.
using System;
using System.IO;
using System.Threading;

static class FakeRelinker
{
    static int Main(string[] args)
    {
        string self = System.Reflection.Assembly.GetExecutingAssembly().Location;
        string dir = Path.GetDirectoryName(self);

        if (!Path.GetFileName(self).Equals("relinker.exe", StringComparison.OrdinalIgnoreCase))
        {
            if (File.Exists(Path.Combine(Environment.CurrentDirectory, "game-hang.txt"))) Thread.Sleep(60000);
            Console.WriteLine(Directory.Exists("app0") ? "app0 visible" : "app0 NOT visible");
            Console.Error.WriteLine("some stderr line");
            return 7;
        }

        string modeFile = Path.Combine(dir, "fake-mode.txt");
        string mode = File.Exists(modeFile) ? File.ReadAllText(modeFile).Trim() : "";
        if (mode == "fail") { Console.Error.WriteLine("error: unsupported syscall in eboot"); return 2; }
        if (mode == "hang") { Thread.Sleep(60000); return 0; }

        string output = args[args.Length - 1];
        string input = args[args.Length - 2];
        Console.WriteLine("Executable file: " + output);
        if (mode == "nooutput") return 0;

        File.Copy(self, output, true);
        string modules = Path.Combine(Path.GetDirectoryName(input), "sce_module");
        string target = Path.Combine(Path.GetDirectoryName(output), "app0", "sce_module");
        Directory.CreateDirectory(target);
        if (Directory.Exists(modules))
            foreach (string m in Directory.GetFiles(modules))
            {
                File.WriteAllText(Path.Combine(target, Path.GetFileName(m) + ".guest.prx"), "converted");
                File.WriteAllText(Path.Combine(target, Path.GetFileName(m)), "new");
            }
        return 0;
    }
}
